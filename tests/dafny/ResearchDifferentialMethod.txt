  // Research differential harness: same edit trace, either persistent resolver cache
  // or fresh ProgramResolver (UseCaching=false) on EVERY document revision.
  public override Task InitializeAsync() {
    return Environment.GetEnvironmentVariable("DAFNY_RESEARCH_MODE") == "fresh"
      ? SetUp(options => options.Set(
          Microsoft.Dafny.LanguageServer.Language.Symbols.DafnyLangSymbolResolver.UseCaching, false))
      : base.InitializeAsync();
  }

  [Theory]
  [InlineData("independent")]
  [InlineData("dependency")]
  [InlineData("shadowing")]
  [InlineData("roster")]
  [InlineData("recovery")]
  public async Task ResearchDifferentialTrace(string scenario) {
    string[] revisions = scenario switch {
      "independent" => new[] {
        "module Alpha {\n function Value(): int { 11 }\n}\nmodule Beta {\n function Value(): int { 21 }\n}\n",
        "module Alpha {\n function Value(): int { 11 }\n}\nmodule Beta {\n function Value(): int { 22 }\n}\n",
        "module Alpha {\n function Value(): int { 11 }\n}\nmodule Beta {\n function Value(): int { 222 }\n}\n",
        "\nmodule Alpha {\n function Value(): int { 11 }\n}\nmodule Beta {\n function Value(): int { 222 }\n}\n"
      },
      "dependency" => new[] {
        "module Provider { function Value(): int { 1 } }\nmodule User { import opened Provider function Result(): int { Value() } }\n",
        "module Provider { function Value(): bool { true } }\nmodule User { import opened Provider function Result(): int { Value() } }\n",
        "module Provider { function Value(): int { 1 } }\nmodule User { import opened Provider function Result(): int { Value() } }\n",
        "module Provider { function Renamed(): int { 1 } }\nmodule User { import opened Provider function Result(): int { Value() } }\n"
      },
      "shadowing" => new[] {
        "module Shared { function Value(): int { 1 } }\nmodule User { import opened Shared function Result(): int { Value() } }\n",
        "module Shared { function Value(): int { 1 } }\nmodule User { import opened Shared function Value(): int { 9 } function Result(): int { Value() } }\n",
        "module Shared { function Value(): int { 1 } }\nmodule User { import opened Shared function Result(): int { Value() } }\n"
      },
      "roster" => new[] {
        "module Alpha { function Value(): int { 11 } }\nmodule Beta { function Value(): int { 21 } }\n",
        "module Alpha { function Value(): int { 11 } }\nmodule Beta { function Value(): int { 21 } }\nmodule Gamma { function Value(): int { 33 } }\n",
        "module Alpha { function Value(): int { 11 } }\nmodule Beta { function Value(): int { 21 } }\nmodule Alpha { function Value(): int { 33 } }\n"
      },
      "recovery" => new[] {
        "module Alpha { function Value(): int { 11 } }\nmodule Beta { function Value(): int { 21 } }\n",
        "module Alpha { function Value(): int { 11 } }\nmodule Beta { function Value(): int { } }\n",
        "module Alpha { function Value(): int { 11 } }\nmodule Beta { function Value(): int { 21 } }\n",
        "module Alpha { function Value(): int { 11 } }\nmodule Beta { function Value(): int { true } }\n"
      },
      _ => throw new ArgumentException(scenario)
    };
    var path = Path.Combine(Directory.GetCurrentDirectory(),
      "Synchronization/TestFiles/research-differential-" + scenario + ".dfy");
    var document = CreateTestDocument(revisions[0], path);
    await client.OpenDocumentAndWaitAsync(document, CancellationToken);
    var records = new System.Collections.Generic.List<object>();
    Diagnostic[] lastPublishedDiagnostics = Array.Empty<Diagnostic>();
    string mode = Environment.GetEnvironmentVariable("DAFNY_RESEARCH_MODE") ?? "unconfigured";
    for (int index = 0; index < revisions.Length; index++) {
      if (index != 0) {
        var oldLines = revisions[index - 1].Split('\n');
        ApplyChange(ref document, new Range(0, 0, oldLines.Length - 1,
          oldLines[oldLines.Length - 1].Length), revisions[index]);
        await client.WaitForNotificationCompletionAsync(document.Uri, CancellationToken);
      }
      // A valid edit may publish no new diagnostics; do not block waiting for one.
      // Preserve the last published diagnostic snapshot when no event is emitted.
      var published = diagnosticsReceiver.GetLatestAndClearQueue(d => d.Uri == document.Uri);
      if (published != null) {
        lastPublishedDiagnostics = published.Diagnostics.ToArray();
      }
      var diagnostics = lastPublishedDiagnostics;
      var normalized = diagnostics.Select(d => new {
        message = d.Message,
        severity = d.Severity?.ToString(),
        code = d.Code.HasValue ? (d.Code.Value.IsString ? d.Code.Value.String : d.Code.Value.Long.ToString()) : null,
        source = d.Source,
        from = new[] { d.Range.Start.Line, d.Range.Start.Character },
        to = new[] { d.Range.End.Line, d.Range.End.Character },
        related = d.RelatedInformation == null ? Array.Empty<string>() :
          d.RelatedInformation.Select(r => r.Message + "@" +
            r.Location.Uri.ToString() + ":" +
            r.Location.Range.Start.Line + ":" + r.Location.Range.Start.Character).ToArray()
      }).OrderBy(d => d.from[0]).ThenBy(d => d.from[1])
        .ThenBy(d => d.message).ToArray();
      records.Add(new { revision = index, source = revisions[index], diagnostics = normalized });
    }
    var output = Environment.GetEnvironmentVariable("DAFNY_RESEARCH_OUTPUT_DIR");
    if (string.IsNullOrWhiteSpace(output)) {
      throw new InvalidOperationException("DAFNY_RESEARCH_OUTPUT_DIR missing");
    }
    Directory.CreateDirectory(output);
    File.WriteAllText(Path.Combine(output, mode + "-" + scenario + ".json"),
      System.Text.Json.JsonSerializer.Serialize(new { mode, scenario, records },
        new System.Text.Json.JsonSerializerOptions { WriteIndented = true }));
    Assert.Equal(revisions.Length, records.Count);
  }
