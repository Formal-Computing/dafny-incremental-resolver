  // Captures post-resolution, post-update diagnostics from a persistent LSP.
  // The fresh condition invokes ProgramResolver on every revision, not a new process per edit.
  public override Task InitializeAsync() {
    return Environment.GetEnvironmentVariable("DAFNY_RESEARCH_TRACE_MODE") == "fresh"
      ? SetUp(options => options.Set(
          Microsoft.Dafny.LanguageServer.Language.Symbols.DafnyLangSymbolResolver.UseCaching, false))
      : base.InitializeAsync();
  }

  [Theory]
  [InlineData("independent")]
  [InlineData("dependency")]
  [InlineData("shadowing")]
  [InlineData("roster")]
  [InlineData("error-recovery")]
  public async Task ResearchDifferentialTrace(string scenario) {
    string[] versions = scenario switch {
      "independent" => new[] {
        "module Alpha {\n  function Value(): int { 11 }\n}\nmodule Beta {\n  function Value(): int { 21 }\n}\n",
        "module Alpha {\n  function Value(): int { 11 }\n}\nmodule Beta {\n  function Value(): int { 22 }\n}\n",
        "module Alpha {\n  function Value(): int { 11 }\n}\nmodule Beta {\n  function Value(): int { 222 }\n}\n",
        "\nmodule Alpha {\n  function Value(): int { 11 }\n}\nmodule Beta {\n  function Value(): int { 222 }\n}\n",
        "module Alpha {\n  function Value(): int { 11 }\n}\nmodule Beta {\n  function Value(): int { 21 }\n}\n"
      },
      "dependency" => new[] {
        "module Provider { function Value(): int { 1 } }\nmodule User { import opened Provider function Result(): int { Value() } }\n",
        "module Provider { function Value(): bool { true } }\nmodule User { import opened Provider function Result(): int { Value() } }\n",
        "module Provider { function Value(): int { 1 } }\nmodule User { import opened Provider function Result(): int { Value() } }\n",
        "module Provider { function Renamed(): int { 1 } }\nmodule User { import opened Provider function Result(): int { Value() } }\n",
        "module Provider { function Value(): int { 1 } }\nmodule User { import opened Provider function Result(): int { Value() } }\n"
      },
      "shadowing" => new[] {
        "module Shared { function Value(): int { 1 } }\nmodule User { import opened Shared function Result(): int { Value() } }\n",
        "module Shared { function Value(): int { 1 } }\nmodule User { import opened Shared function Value(): int { 9 } function Result(): int { Value() } }\n",
        "module Shared { function Value(): int { 1 } }\nmodule User { import opened Shared function Result(): int { Value() } }\n"
      },
      "roster" => new[] {
        "module Alpha { function Value(): int { 11 } }\nmodule Beta { function Value(): int { 21 } }\n",
        "module Alpha { function Value(): int { 11 } }\nmodule Beta { function Value(): int { 21 } }\nmodule Gamma { function Value(): int { 33 } }\n",
        "module Alpha { function Value(): int { 11 } }\nmodule Beta { function Value(): int { 21 } }\nmodule Alpha { function Value(): int { 33 } }\n",
        "module Alpha { function Value(): int { 11 } }\nmodule Beta { function Value(): int { 21 } }\n"
      },
      "error-recovery" => new[] {
        "module Alpha { function Value(): int { 11 } }\nmodule Beta { function Value(): int { 21 } }\n",
        "module Alpha { function Value(): int { 11 } }\nmodule Beta { function Value(): int { } }\n",
        "module Alpha { function Value(): int { 11 } }\nmodule Beta { function Value(): int { 21 } }\n",
        "module Alpha { function Value(): int { 11 } }\nmodule Beta { function Value(): int { true } }\n",
        "module Alpha { function Value(): int { 11 } }\nmodule Beta { function Value(): int { 21 } }\n"
      },
      _ => throw new ArgumentException("Unknown scenario " + scenario)
    };
    var path = Path.Combine(Directory.GetCurrentDirectory(),
      "Synchronization/TestFiles/research-differential-" + scenario + ".dfy");
    var document = CreateTestDocument(versions[0], path);
    await client.OpenDocumentAndWaitAsync(document, CancellationToken);
    var trace = new System.Collections.Generic.List<object>();
    string mode = Environment.GetEnvironmentVariable("DAFNY_RESEARCH_TRACE_MODE") ?? "unspecified";
    for (int index = 0; index < versions.Length; index++) {
      if (index > 0) {
        var oldLines = versions[index - 1].Split('\n');
        ApplyChange(ref document, new Range(0, 0, oldLines.Length - 1,
          oldLines[oldLines.Length - 1].Length), versions[index]);
      }
      // GetLastDiagnostics waits for this exact document revision to finish processing,
      // and throws on an internal resolver exception.
      var diagnostics = await GetLastDiagnostics(document, DiagnosticSeverity.Hint);
      var stable = diagnostics.Select(d => new {
        severity = d.Severity?.ToString(),
        code = d.Code?.ToString(),
        message = d.Message,
        source = d.Source,
        start = new[] {d.Range.Start.Line, d.Range.Start.Character},
        end = new[] {d.Range.End.Line, d.Range.End.Character},
        related = (d.RelatedInformation ?? new())
          .Select(r => new {
            uri = r.Location.Uri.ToString().Split('/').Last(),
            message = r.Message,
            start = new[] {r.Location.Range.Start.Line, r.Location.Range.Start.Character},
            end = new[] {r.Location.Range.End.Line, r.Location.Range.End.Character}
          }).OrderBy(r => r.message).ToArray()
      }).OrderBy(d => d.start[0]).ThenBy(d => d.start[1])
        .ThenBy(d => d.code).ThenBy(d => d.message).ToArray();
      trace.Add(new {revision = index, source = versions[index], diagnostics = stable});
    }
    var outputDirectory = Environment.GetEnvironmentVariable("DAFNY_RESEARCH_OUTPUT_DIR");
    if (string.IsNullOrEmpty(outputDirectory)) {
      throw new InvalidOperationException("DAFNY_RESEARCH_OUTPUT_DIR must be set");
    }
    Directory.CreateDirectory(outputDirectory);
    var json = System.Text.Json.JsonSerializer.Serialize(new {
      mode, scenario, dafnyRevision = "de36f2ec1f9c530080eff7929ae28229f45da069", trace
    }, new System.Text.Json.JsonSerializerOptions {WriteIndented = true});
    File.WriteAllText(Path.Combine(outputDirectory, mode + "-" + scenario + ".json"), json);
    Assert.Equal(versions.Length, trace.Count);
  }
